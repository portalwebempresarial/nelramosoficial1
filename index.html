2 Conector Google Apps Script (Cero mantenimiento)

Ve a script.google.com, crea un proyecto nuevo, pega este código, reemplaza TU_ID_CARPETA_DRIVE y dale en Implementar > Nueva implementación > Aplicación web (Acceso: Cualquier persona). 

function doGet(e) {
  // Pega aquí el ID de tu carpeta pública de Google Drive
  var FOLDER_ID = e.parameter.folderId || "TU_ID_CARPETA_DRIVE";
  var folder = DriveApp.getFolderById(FOLDER_ID);
  var files = folder.getFiles();
  var catalog = [];

  while (files.hasNext()) {
    var file = files.next();
    var mime = file.getMimeType();
    if (mime.indexOf("image/") === 0) {
      catalog.push({
        id: file.getId(),
        name: file.getName(),
        updated: file.getLastUpdated().getTime()
      });
    }
  }

  return ContentService
    .createTextOutput(JSON.stringify({ status: "ok", items: catalog }))
    .setMimeType(ContentService.MimeType.JSON);
}
